\section{Boundary Conditions}
We have the conservation of charge on the interfaces\begin{equation}
    \frac{\partial\rho}{\partial{t}}=-\nabla\cdot\mathbf{j}
\end{equation}and the Ohm's law\begin{equation}\renewcommand{\arraystretch}{1.3}
    \mathbf{j}=\sigma\mathbf{E}=\begin{pmatrix}
    0 & \sigma_{xy}\\
    -\sigma_{xy} & 0\\
\end{pmatrix}\begin{pmatrix}
E_x \\
E_y \\
\end{pmatrix}=\begin{pmatrix}
\sigma_{xy}E_y \\
-\sigma_{xy}E_x \\
\end{pmatrix}
\end{equation}thus\begin{equation}
    \frac{\partial\rho}{\partial{t}}=-\sigma_{xy}(\frac{\partial{E_y}}{\partial{x}}-\frac{\partial{E_x}}{\partial{y}})=\sigma_{xy}\frac{\partial{B_z}}{\partial{t}}\Rightarrow{\rho=\sigma_{xy}B_z}
\end{equation}
Plugging this into the Maxwell equations, we have\begin{equation}
    \nabla\cdot\mathbf{E}=\frac{\sigma_{xy}}{\varepsilon_0}B_z
\end{equation}\begin{equation}
    \nabla\times\mathbf{E}=-\frac{\partial\mathbf{B}}{\partial{t}}
\end{equation}\begin{equation}
    \nabla\cdot\mathbf{B}=0
\end{equation}\begin{equation}
    \nabla\times\mathbf{H}=\mathbf{j}+\varepsilon\frac{\partial\mathbf{E}}{\partial{t}}=\sigma\mathbf{E}\delta(z-z_0)+\varepsilon\frac{\partial{\mathbf{E}}}{\partial{t}}
\end{equation}
We use the superscript $R$ and $L$ to distinguish right and left side of the interface and $+(-)$ to denote the direction of light propagation. By approximating the relative permeability constant in each region to 1, the electric and magnetic fields are given by (notice the wave vector is no longer perpendicular to the surface. But for simplicity, we just let the $k_y$ of the incident light be 0, so that the $k_y$ of both transmitted light and reflected light vanish)\begin{equation}
    \mathbf{B}^{+L}=\frac{n_L}{c}[-{E}_y^{+L}\cos\theta\hat{x}+(E_x^{+L}\cos\theta-E_z^{+L}\sin\theta)\hat{y}+E_y^{+L}\sin\theta\hat{z}]
\end{equation}\begin{equation}
    \mathbf{B}^{-L}=\frac{n_L}{c}[{E}_y^{-L}\cos\theta\hat{x}+(-E_x^{-L}\cos\theta-E_z^{-L}\sin\theta)\hat{y}+E_y^{-L}\sin\theta\hat{z}]
\end{equation}\begin{equation}
    \mathbf{B}^{+R}=\frac{n_R}{c}[-{E}_y^{+R}\cos\phi\hat{x}+(E_x^{+R}\cos\phi-E_z^{+R}\sin\phi)\hat{y}+E_y^{+R}\sin\phi\hat{z}]
\end{equation}\begin{equation}
    \mathbf{B}^{-R}=\frac{n_R}{c}[{E}_y^{-R}\cos\phi\hat{x}+(-E_x^{-R}\cos\phi-E_z^{-R}\sin\phi)\hat{y}+E_y^{-R}\sin\phi\hat{z}]
\end{equation}
where $\theta$ and $\phi$ represent the angle between the incident light, transmitted light and the interface normal, respectively, and they're connected through the law of refraction\begin{equation}
    n_L\sin\theta=n_R\sin\phi
\end{equation} Moreover, from the Maxwell equations we can derive the boundary conditions for $\mathbf{E}$ and $\mathbf{B}$\begin{equation}
    E_z^R-E_z^L=\frac{\sigma_{xy}}{\varepsilon_0}B_z^L
\end{equation}\begin{equation}
    E_x^R=E_x^L
\end{equation}\begin{equation}
    E_y^R=E_y^L
\end{equation}\begin{equation}
    B_x^L-B_x^R=\mu_0\sigma_{xy}E_x
\end{equation}\begin{equation}
    B_y^L-B_y^R=\mu_0\sigma_{xy}E_y
\end{equation}\begin{equation}
    B_z^L=B_z^R
\end{equation}
All components of electric field and magnetic field is unknown, which add up to 24 unknowns. To derive the transfer matrix, we set the incident components as known. Then the number of equations (18 equations) matches the unknowns.
\section{Transfer Matrices}
The boundary conditions obtained in section 1 are used in this section to derive the transfer matrix when light propagates through the interfaces. Since the z-component of electric field has a strict geometric constraint, or more precisely, uniquely determined by the incident angle and the x-component of electric field (two of our controllable variables). Moreover, through calculation, we find that the tangential components on the right side of the interface are uniquely determined by that on the left side. So if we just concern about the tangential components, the transfer matrix is still $4\times4$.\begin{equation}\renewcommand{\arraystretch}{1.7}
\begin{pmatrix}
    E_x^{+R}\\
    E_y^{+R}\\
    E_x^{-R}\\
    E_y^{-R}
    \end{pmatrix}=\mathrm{T^I}\begin{pmatrix}
    E_x^{+L}\\
    E_y^{+L}\\
    E_x^{-L}\\
    E_y^{-L}
    \end{pmatrix}
\end{equation}
where the transfer matrix $\mathrm{T^I}(n_L,n_R,\sigma_{xy},\theta)$ are displayed as\begin{equation}\renewcommand{\arraystretch}{1.9}
    \mathrm{T^I}=\frac12\begin{pmatrix}
    1+\frac{n_L\cos\theta}{n_R\cos\phi} & \frac{2\alpha\sigma_{xy}(n_L^2\sin^2\theta-1)}{n_R\cos\phi} & 1-\frac{n_L\cos\theta}{n_R\cos\phi} & \frac{2\alpha\sigma_{xy}(n_L^2\sin^2\theta-1)}{n_R\cos\phi}\\
    \frac{2\alpha\sigma_{xy}}{n_R\cos\phi} & 1+\frac{n_L\cos\theta}{n_R\cos\phi} & \frac{2\alpha\sigma_{xy}}{n_R\cos\phi} & 1-\frac{n_L\cos\theta}{n_R\cos\phi}\\
    1-\frac{n_L\cos\theta}{n_R\cos\phi} & -\frac{2\alpha\sigma_{xy}(n_L^2\sin^2\theta-1)}{n_R\cos\phi} & 1+\frac{n_L\cos\theta}{n_R\cos\phi} & -\frac{2\alpha\sigma_{xy}(n_L^2\sin^2\theta-1)}{n_R\cos\phi}\\
    -\frac{2\alpha\sigma_{xy}}{n_R\cos\phi} & 1-\frac{n_L\cos\theta}{n_R\cos\phi} & -\frac{2\alpha\sigma_{xy}}{n_R\cos\phi} & 1+\frac{n_L\cos\theta}{n_R\cos\phi}
    \end{pmatrix}
\end{equation}
The Hall conductivity $\sigma_{xy}$ is in a unit of $\frac{e^2}{h}$. Then, considering the interference of the light due to multiple reflections inside the film, the transfer matrix for the propagation in the topological insulator bulk can be regarded as an identity matrix since we're calculating all these things based on the long-wavelength approximation. The total transfer matrix is given by the product of the transfer matrices\begin{equation}
    \mathrm{T^{total}}=\mathrm{T^I}(n_f,n_s,\sigma_{xy}^b,\phi)\mathrm{T^I}(n_v,n_f,\sigma_{xy}^t,\theta)
\end{equation}\begin{equation}\renewcommand{\arraystretch}{2.1}
    \mathrm{T^{total}}=\frac12\begin{pmatrix}
    1+\frac{n_v\cos\theta}{n_s\cos\gamma} & \frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)(n_v^2\sin^2\theta-1)}{n_s\cos\gamma} & 1-\frac{n_v\cos\theta}{n_s\cos\gamma} & \frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)(n_v^2\sin^2\theta-1)}{n_s\cos\gamma}\\
    \frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)}{n_s\cos\gamma} & 1+\frac{n_v\cos\theta}{n_s\cos\gamma} & \frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)}{n_s\cos\gamma} & 1-\frac{n_v\cos\theta}{n_s\cos\gamma}\\
    1-\frac{n_v\cos\theta}{n_s\cos\gamma} & -\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)(n_v^2\sin^2\theta-1)}{n_s\cos\gamma} & 1+\frac{n_v\cos\theta}{n_s\cos\gamma} & -\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)(n_v^2\sin^2\theta-1)}{n_s\cos\gamma}\\
    -\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)}{n_s\cos\gamma} & 1-\frac{n_v\cos\theta}{n_s\cos\gamma} & -\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)}{n_s\cos\gamma} & 1+\frac{n_v\cos\theta}{n_s\cos\gamma}
    \end{pmatrix}
\end{equation}
where the angles follow the law of refraction\begin{equation}
    n_v\sin\theta=n_f\sin\phi=n_s\sin\gamma
\end{equation}
We can do a little estimation here. The thickness of the substrate is about 300 $\mu$m, but the duration of laser pulse is merely 100 fs. So the reflected wave from the back interface of the substrate is negligible when calculating the Faraday rotation angle. In this way, the transmitted and reflected light are related by the following equation\begin{equation}\renewcommand{\arraystretch}{1.7}
\begin{pmatrix}
    E_x^t\\
    E_y^t\\
    0\\
    0
    \end{pmatrix}=\mathrm{T^{total}}\begin{pmatrix}
    E_x^i\\
    0\\
    E_x^r\\
    E_y^r
    \end{pmatrix}
\end{equation}
That is\begin{equation}
    2E_x^t=(1+\frac{n_v\cos\theta}{n_s\cos\gamma})E_x^i+(1-\frac{n_v\cos\theta}{n_s\cos\gamma})E_x^r+\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)(n_v^2\sin^2\theta-1)}{n_s\cos\gamma}E_y^r
\end{equation}\begin{equation}
    2E_y^t=\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)}{n_s\cos\gamma}E_x^i+\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)}{n_s\cos\gamma}E_x^r+(1-\frac{n_v\cos\theta}{n_s\cos\gamma})E_y^r
\end{equation}\begin{equation}
    0=(1-\frac{n_v\cos\theta}{n_s\cos\gamma})E_x^i+(1+\frac{n_v\cos\theta}{n_s\cos\gamma})E_x^r-\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)(n_v^2\sin^2\theta-1)}{n_s\cos\gamma}E_y^r
\end{equation}\begin{equation}
    0=-\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)}{n_s\cos\gamma}E_x^i-\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)}{n_s\cos\gamma}E_x^r+(1+\frac{n_v\cos\theta}{n_s\cos\gamma})E_y^r
\end{equation}
We get the Faraday rotation angle as\begin{equation}
    \tan\theta_F=\frac{E_y^t}{E_x^t}=\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)}{n_s\cos\gamma+n_v\cos\theta}
\end{equation}
The influence on Faraday rotation angle caused by the back interface is also negligible due to our previous estimation since we're focusing only on the amplitudes of $E_y$ and $E_x$. The same applies to the Kerr rotation where the light is reflected at the substrate interface after experiencing the Faraday rotation. In this case, the total transfer matrix is rewritten as\begin{equation}
    \mathrm{T^{total}}=\mathrm{T^I}(n_f,n_v,\sigma_{xy}^t,\phi)\mathrm{T^I}(n_s,n_f,\sigma_{xy}^b,\gamma)
\end{equation}\begin{equation}\renewcommand{\arraystretch}{2.1}
    \mathrm{T^{total}}=\frac12\begin{pmatrix}
    1+\frac{n_s\cos\gamma}{n_v\cos\theta} & \frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)(n_s^2\sin^2\gamma-1)}{n_v\cos\theta} & 1-\frac{n_s\cos\gamma}{n_v\cos\theta} & \frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)(n_s^2\sin^2\gamma-1)}{n_v\cos\theta}\\
    \frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)}{n_v\cos\theta} & 1+\frac{n_s\cos\gamma}{n_v\cos\theta} & \frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)}{n_v\cos\theta} & 1-\frac{n_s\cos\gamma}{n_v\cos\theta}\\
    1-\frac{n_s\cos\gamma}{n_v\cos\theta} & -\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)(n_s^2\sin^2\gamma-1)}{n_v\cos\theta} & 1+\frac{n_s\cos\gamma}{n_v\cos\theta} & -\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)(n_s^2\sin^2\gamma-1)}{n_v\cos\theta}\\
    -\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)}{n_v\cos\theta} & 1-\frac{n_s\cos\gamma}{n_v\cos\theta} & -\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)}{n_v\cos\theta} & 1+\frac{n_s\cos\gamma}{n_v\cos\theta}
    \end{pmatrix}
\end{equation}In our measurement, the Kerr and Faraday rotation are combined together when we're obtaining the different components of the delayed second pulse. So here we shall still let $\theta$ to be the only variable. The transmitted and reflected light are related by following four equations\begin{equation}
    2E_x^t=(1+\frac{n_s\cos\gamma}{n_v\cos\theta})E_x^i+(1-\frac{n_s\cos\gamma}{n_v\cos\theta})E_x^r+\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)(n_s^2\sin^2\gamma-1)}{n_v\cos\theta}E_y^r
\end{equation}\begin{equation}
    2E_y^t=\frac{2\alpha(\sigma_{xy}^t+\sigma_xy^{b})}{n_v\cos\theta}E_x^i+\frac{2\alpha(\sigma_{xy}^t+\sigma_xy^{b})}{n_v\cos\theta}E_x^r+(1-\frac{n_s\cos\gamma}{n_v\cos\theta})E_y^r
\end{equation}\begin{equation}
    0=(1-\frac{n_s\cos\gamma}{n_v\cos\theta})E_x^i+(1+\frac{n_s\cos\gamma}{n_v\cos\theta})E_x^r-\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)(n_s^2\sin^2\gamma-1)}{n_v\cos\theta}E_y^r
\end{equation}\begin{equation}
    0=-\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)}{n_v\cos\theta}E_x^i-\frac{2\alpha(\sigma_{xy}^t+\sigma_{xy}^b)}{n_v\cos\theta}E_x^r+(1+\frac{n_s\cos\gamma}{n_v\cos\theta})E_y^r
\end{equation}
We get the Kerr rotation angle as\begin{equation}
    \tan\theta_K=\frac{4\alpha{n_s}(\sigma_{xy}^t+\sigma_{xy}^b)\cos\gamma}{n_s^2\cos^2\gamma-n_v^2\cos^2\theta+4\alpha^2(\sigma_{xy}^t+\sigma_{xy}^b)^2(n_s^2\sin^2\gamma-1)}
\end{equation}
or a more simplified form\begin{equation}
    \tan\theta_K=\frac{4\alpha{n_s}(\sigma_{xy}^t+\sigma_{xy}^b)\cos\gamma}{n_s^2-n_v^2+4\alpha^2(\sigma_{xy}^t+\sigma_{xy}^b)^2(n_v^2\sin^2\theta-1)}
\end{equation}
\section{Discussion}
If we're talking about the Streda scenario, then both $\sigma_{xy}^t$ and $\sigma_{xy}^b$ are $\frac12(\frac{e^2}{h})$, which means (refractive index for vacuum is 1)
\begin{equation}
    \tan\theta_F=\frac{2\alpha}{n_s\cos\gamma+\cos\theta}=\frac{2\alpha}{\sqrt{n_s^2-\sin^2\theta}+\cos\theta}
\end{equation}\begin{equation}
    \tan\theta_K=\frac{4\alpha{n_s}\cos\gamma}{n_s^2-1-4\alpha^2\cos^2\theta}=\frac{4\alpha\sqrt{n_s^2-\sin^2\theta}}{n_s^2-1-4\alpha^2\cos^2\theta}
\end{equation}
Since the fine structure constant $\alpha\approx\frac{1}{137}$, so it's quite reasonable to validate the following approximation\begin{equation}
    \theta_F\approx\arctan{\frac{2\alpha}{\sqrt{n_s^2-\sin^2\theta}+\cos\theta}}
\end{equation}\begin{equation}
    \theta_K\approx\arctan{\frac{4\alpha\sqrt{n_s^2-\sin^2\theta}}{n_s^2-1}}
\end{equation}
While these two quantities depend on the refractive index of the substrate, we can extract $\alpha$ by combining them\begin{equation}
    \alpha=\frac{\tan\theta_F\tan\theta_K-\tan^2\theta_F}{\tan\theta_K-2\tan\theta_F}\cos\theta
\end{equation}
If we further assume the incident angle is small enough to validate the approximation $\sin\theta\approx\theta$, then we have\begin{equation}
    \theta_F=\arctan{\frac{2\alpha}{n_s+1}}+\mathcal{O}(\alpha\theta^2)
\end{equation}\begin{equation}
    \theta_K=\arctan{\frac{4\alpha{n_s}}{n_s^2-1}}+\mathcal{O}(\alpha\theta^2)
\end{equation}
